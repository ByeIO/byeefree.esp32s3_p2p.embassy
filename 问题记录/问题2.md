// 导入需要的库
use futures_lite::future::block_on;  // 用于阻塞执行异步任务
use nusb::{
    transfer::{ControlIn, ControlOut, ControlType, Recipient},
    MaybeFuture,
};

fn main() {
    // 初始化日志记录器
    env_logger::init();
    
    // 枚举USB设备并等待结果
    let di = nusb::list_devices()
        .wait()  // 阻塞等待设备枚举完成
        .unwrap()
        // 查找特定VID/PID的设备(ESP32-S2示例)
        .find(|d| d.vendor_id() == 0x303a && d.product_id() == 0x1001)
        .expect("设备未连接");  // 若未找到则panic

    println!("发现设备: {di:?}");  // 打印设备信息

    // 打开USB设备
    let device = di.open().wait().unwrap();

    // Linux/macOS可以直接发送控制传输而无需声明接口
    #[cfg(any(target_os = "linux", target_os = "macos"))]
    {
        // 发送控制输出传输（主机到设备）
        let result = block_on(device.control_out(ControlOut {
            control_type: ControlType::Vendor,  // 供应商自定义类型
            recipient: Recipient::Device,       // 传输目标是设备本身
            request: 0x81,       // 请求码
            value: 0x9999,       // 参数值
            index: 0x9999,       // 索引值
            data: &[1, 2, 3, 4], // 要发送的数据
        }));
        println!("控制输出结果: {result:?}");

        // 发送控制输入传输（设备到主机）
        let result = block_on(device.control_in(ControlIn {
            control_type: ControlType::Vendor,
            recipient: Recipient::Device,
            request: 0x81,
            value: 0x9999,
            index: 0x9999,
            length: 256,         // 期望接收的数据长度
        }));
        println!("控制输入结果: {result:?}");
    }

    // Windows需要声明接口后才能操作
    // 声明接口0（通常第一个接口）
    let interface = device.claim_interface(0).wait().unwrap();

    // 通过接口发送控制输出传输
    let result = block_on(interface.control_out(ControlOut {
        control_type: ControlType::Vendor,
        recipient: Recipient::Device,
        request: 0x81,
        value: 0x9999,
        index: 0x9999,
        data: &[1, 2, 3, 4],
    }));
    println!("接口控制输出结果: {result:?}");

    // 通过接口发送控制输入传输
    let result = block_on(interface.control_in(ControlIn {
        control_type: ControlType::Vendor,
        recipient: Recipient::Device,
        request: 0x81,
        value: 0x9999,
        index: 0x9999,
        length: 256,
    }));
    println!("接口控制输入结果: {result:?}");
}

代码关键点说明：

1. **设备枚举**：通过VID/PID(0x303A/0x1001)查找特定USB设备，这是ESP32-S2的典型标识符

2. **跨平台处理**：
   - Linux/macOS可以直接发送控制传输
   - Windows需要先声明接口（claim_interface）

3. **控制传输参数**：
   - ControlType::Vendor：表示使用供应商自定义的传输类型
   - Recipient::Device：指定传输目标为整个设备
   - request/value/index：由设备固件定义的特定参数

4. **异步操作**：
   - 使用block_on将异步操作转换为同步执行
   - .wait()是nusb库提供的便捷异步等待方法

5. **数据传输类型**：
   - control_out：主机发送数据到设备（写操作）
   - control_in：主机从设备读取数据（读操作）

此代码演示了通过USB进行供应商自定义控制传输的标准流程，适用于与嵌入式设备（如ESP32系列）进行底层通信的场景。

#![allow(unused_imports)]
use nusb::{
    transfer::{ControlIn, ControlOut, ControlType, Endpoint, Interface, Recipient},
    Error as NusbError,
};
use std::time::Duration;

// CDC类特定请求类型（参考USB CDC规范）
const SET_LINE_CODING: u8 = 0x20;
const GET_LINE_CODING: u8 = 0x21;
const SET_CONTROL_LINE_STATE: u8 = 0x22;

/// 自定义USB错误类型
#[derive(Debug)]
pub enum UsbError {
    WouldBlock,
    Other(NusbError),
}

impl From<NusbError> for UsbError {
    fn from(e: NusbError) -> Self {
        match e {
            NusbError::Timeout => UsbError::WouldBlock,
            _ => UsbError::Other(e),
        }
    }
}

/// USB-CDC 串口设备结构体
pub struct SerialPort {
    interface: Interface,
    bulk_in: Endpoint,
    bulk_out: Endpoint,
}

impl SerialPort {
    /// 创建并初始化USB-CDC设备
    pub fn new() -> Result<Self, UsbError> {
        // 查找指定VID/PID的设备
        let di = nusb::list_devices()
            .wait()
            .map_err(UsbError::from)?
            .find(|d| d.vendor_id() == 0x303a && d.product_id() == 0x1001)
            .expect("设备未连接");

        // 打开设备并声明接口
        let device = di.open().wait().map_err(UsbError::from)?;
        let interface = device.claim_interface(0).wait().map_err(UsbError::from)?;

        // 配置串口参数
        configure_serial_port(&interface)?;

        // 获取数据端点（假设端点号：输出0x01，输入0x81）
        let bulk_out = interface.bulk_out(0x01).map_err(UsbError::from)?;
        let bulk_in = interface.bulk_in(0x81).map_err(UsbError::from)?;

        // 设置端点超时时间为0（非阻塞模式）
        bulk_in.set_timeout(Duration::from_secs(0));
        bulk_out.set_timeout(Duration::from_secs(0));

        Ok(Self {
            interface,
            bulk_in,
            bulk_out,
        })
    }

    /// 从设备读取数据（非阻塞）
    pub fn read(&mut self, buf: &mut [u8]) -> Result<usize, UsbError> {
        self.bulk_in.read(buf).wait().map_err(UsbError::from)
    }

    /// 向设备写入数据（非阻塞）
    pub fn write(&mut self, data: &[u8]) -> Result<usize, UsbError> {
        self.bulk_out.write(data).wait().map_err(UsbError::from)
    }
}

/// 配置串口参数（波特率、数据位、停止位等）
fn configure_serial_port(interface: &Interface) -> Result<(), UsbError> {
    // 设置线路编码（115200 8N1）
    let line_coding = [
        0x00, 0xC2, 0x01, 0x00, // 115200 小端格式（0x0001C200）
        0x00,                   // 1位停止位
        0x00,                   // 无校验
        0x08,                   // 8位数据位
    ];

    // 发送SET_LINE_CODING控制请求
    interface
        .control_out(ControlOut {
            control_type: ControlType::Class,
            recipient: Recipient::Interface,
            request: SET_LINE_CODING,
            value: 0x0000,
            index: 0x00,
            data: &line_coding,
        })
        .wait()
        .map_err(UsbError::from)?;

    // 启用DTR（数据终端就绪）
    interface
        .control_out(ControlOut {
            control_type: ControlType::Class,
            recipient: Recipient::Interface,
            request: SET_CONTROL_LINE_STATE,
            value: 0x01,
            index: 0x00,
            data: &[],
        })
        .wait()
        .map_err(UsbError::from)?;

    Ok(())
}

fn main() {
    env_logger::init();

    // 初始化USB-CDC设备
    let mut serial = SerialPort::new().expect("无法打开串口设备");

    loop {
        // 读取缓冲区（64字节）
        let mut buf = [0u8; 64];
        
        // 非阻塞读取
        match serial.read(&mut buf) {
            Ok(count) if count > 0 => {
                println!("收到 {} 字节: {:?}", count, &buf[..count]);
            }
            Err(UsbError::WouldBlock) => {} // 无数据不处理
            Err(e) => eprintln!("读取错误: {:?}", e),
        }

        // 发送笑脸符号 ":)"
        match serial.write(&[0x3a, 0x29]) {
            Ok(count) => println!("发送 {} 字节", count),
            Err(UsbError::WouldBlock) => {} // 缓冲区满不处理
            Err(e) => eprintln!("写入错误: {:?}", e),
        }

        // 降低CPU占用（100ms间隔）
        std::thread::sleep(Duration::from_millis(100));
    }
}

## 回答
USB的**Bulk传输**和**Endpoint**是USB协议中两个核心概念，分别对应数据传输方式与通信终端点的定义。以下是两者的详细解释及关联：

---

### 一、Endpoint（端点）
#### 定义与功能
- **基本概念**：  
  Endpoint是USB设备内部用于数据传输的逻辑终端点，每个设备通过端点与主机进行通信。每个端点具有唯一的地址（由4位ENDP字段定义）和方向（IN或OUT）。例如，IN端点用于设备向主机发送数据，OUT端点用于主机向设备发送数据[4](@ref)[29](@ref)。
  
- **端点类型**：  
  USB规范定义了四种端点类型，对应四种数据传输方式：
  1. **控制传输端点（Control Endpoint）**：用于设备配置、命令传输等，所有USB设备必须包含端点0（默认控制端点）[4](@ref)[29](@ref)。
  2. **中断传输端点（Interrupt Endpoint）**：用于小数据量、低延迟的传输（如鼠标、键盘）[29](@ref)[36](@ref)。
  3. **批量传输端点（Bulk Endpoint）**：用于大量数据的高可靠性传输（如U盘、打印机）[6](@ref)[29](@ref)。
  4. **等时传输端点（Isochronous Endpoint）**：用于实时性要求高但允许数据丢失的场景（如摄像头、音频设备）[29](@ref)[36](@ref)。

- **端点数量与限制**：  
  - 每个设备最多支持32个端点（16个IN + 16个OUT），低速设备限制更严格（如仅支持端点0和一个数据端点）[29](@ref)[27](@ref)。
  - 端点0是唯一双向端点，用于设备枚举和初始配置[4](@ref)[27](@ref)。

---

### 二、Bulk传输（批量传输）
#### 定义与特点
- **基本概念**：  
  Bulk传输是USB四种传输类型之一，专为**大量数据的高可靠性传输**设计，适用于对实时性要求低但数据完整性要求高的场景（如文件传输、存储设备）[6](@ref)[36](@ref)。

- **核心特性**：
  1. **可靠性**：通过CRC校验和自动重传机制确保数据无误[6](@ref)[29](@ref)。
  2. **无带宽保证**：仅在总线空闲时传输，优先级低于中断和同步传输[6](@ref)[36](@ref)。
  3. **方向灵活**：支持IN（设备→主机）和OUT（主机→设备）双向传输[6](@ref)[29](@ref)。
  4. **数据包大小**：不同速度设备的最大包大小不同（全速64字节、高速512字节、超高速1024字节）[6](@ref)[29](@ref)。

#### 应用场景
- **大容量存储设备**（U盘、移动硬盘）。
- **打印机/扫描仪**：需传输大量打印或扫描数据。
- **数据采集设备**：如传感器批量上传数据[36](@ref)[47](@ref)。

---

### 三、Bulk与Endpoint的关系
1. **Bulk端点的实现**：  
   Bulk传输通过**Bulk端点**实现，设备需在描述符中声明端点的类型为Bulk，并指定方向（IN/OUT）[29](@ref)[36](@ref)。
   
2. **虚拟管线（Virtual Pipe）**：  
   主机与Bulk端点的通信通过虚拟管线完成，这种逻辑连接支持数据流的稳定传输[4](@ref)[29](@ref)。

3. **事务流程**：  
   - **Bulk OUT传输**：主机发送OUT令牌包→数据包→设备返回ACK/NAK响应[6](@ref)[36](@ref)。
   - **Bulk IN传输**：主机发送IN令牌包→设备返回数据包→主机确认[6](@ref)[36](@ref)。

---

### 四、总结对比
| **特性**         | **Endpoint（端点）**                         | **Bulk传输**                              |
|------------------|---------------------------------------------|-------------------------------------------|
| **本质**         | 通信的终端点（逻辑地址）                      | 一种数据传输方式                          |
| **类型**         | 控制、中断、批量、等时四类                    | 基于批量端点的传输模式                    |
| **核心功能**     | 定义数据流向和通信接口                        | 提供高可靠性的大数据量传输                |
| **典型应用**     | 所有USB通信的基础                             | 存储设备、打印机等场景                    |

通过上述分析可见，**Endpoint是USB通信的基础单元**，而**Bulk传输则是基于特定类型端点的数据传输策略**，两者共同支撑了USB协议的高效性和灵活性。

## 回答
以下是全速USB设备（以CDC-ACM类为例）的通信流程详解，结合USB协议规范与工程实践：

---

### **一、设备插入与枚举阶段**
1. **设备检测与复位**  
   当USB设备插入主机时，根集线器检测到端口状态变化，向主机报告新设备插入。主机会发送**总线复位信号**（Reset），设备进入默认状态（地址0），并响应控制端点（Endpoint 0）的通信[3](@ref)。

2. **地址分配**  
   主机通过控制端点发送**SET_ADDRESS**请求（标准设备请求），为设备分配唯一地址（如地址25）。此后设备使用新地址通信，完成初步身份标识[3](@ref)[15](@ref)。

3. **描述符获取**  
   - **设备描述符**：主机通过`GET_DESCRIPTOR`请求获取设备的基本信息（如USB版本、设备类、VID/PID等），例如返回数据：`12 01 00 02 02 00 00 40...`，表明支持USB 2.0、CDC类[15](@ref)。  
   - **配置描述符**：主机进一步获取配置描述符集合，包含接口、端点和CDC特定描述符（如IAD、Header、ACM功能描述符）。例如，配置描述符可能包含两个接口（控制接口+数据接口）和三个端点（控制端点、Bulk IN/OUT端点）[5](@ref)[15](@ref)[50](@ref)。

4. **字符串描述符与语言ID**  
   主机获取厂商、产品名称等字符串描述符，并需先获取语言ID描述符（如0x0409表示英文）[15](@ref)。

---

### **二、配置与初始化阶段**
1. **设置配置**  
   主机发送**SET_CONFIGURATION**请求激活设备的某个配置（通常为配置1）。设备完成端点初始化，并注册CDC类处理函数，进入可用状态[5](@ref)[15](@ref)。

2. **CDC类特定请求处理**  
   - **控制接口（CCI）**：通过控制端点处理类特定请求，例如：  
     - **SetLineCoding**：设置波特率（如115200）、数据位（8位）、停止位（1位）等参数。  
     - **GetLineCoding**：主机获取当前串口配置。  
     - **SetControlLineState**：通知设备串口状态（如DTR/RTS信号）[15](@ref)[50](@ref)。  
   - **中断端点（可选）**：用于异步事件通知（如数据就绪信号）[5](@ref)。

---

### **三、数据传输阶段**
1. **数据接口（DCI）通信**  
   - **Bulk OUT端点**：主机通过OUT端点发送数据（如串口发送指令），设备接收后存入缓冲区。  
   - **Bulk IN端点**：设备通过IN端点主动上传数据（如传感器数据），主机轮询或由设备触发传输[5](@ref)[50](@ref)。  
   - **全速USB特性**：Bulk端点最大包大小为64字节，需通过多次传输完成大数据量通信[3](@ref)[15](@ref)。

2. **数据流控制**  
   - **协议层**：CDC-ACM通过虚拟串口协议模拟UART通信，主机应用层无需感知USB底层细节。
   - **错误处理**：CRC校验失败或超时会触发重传，确保数据完整性[5](@ref)[50](@ref)。

---

### **四、典型数据包示例**
1. **主机发送数据（Bulk OUT）**  
   ```hex
   OUT令牌包 → 数据包（如"123456"） → 设备ACK响应
   ```
2. **设备发送数据（Bulk IN）**  
   ```hex
   IN令牌包 → 设备数据包（如传感器数据） → 主机ACK响应
   ```

---

### **五、关键描述符结构**
| **描述符类型**       | **作用与示例**                                                                 |
|----------------------|-------------------------------------------------------------------------------|
| **设备描述符**       | 声明设备类（0x02为CDC）、VID/PID、端点0最大包大小（64字节）[15](@ref)             |
| **配置描述符集合**   | 包含IAD（联合接口描述符）、CCI接口（端点2为中断IN）、DCI接口（批量IN/OUT端点）[15](@ref)[50](@ref) |
| **功能描述符**       | 如Header描述符（标识CDC版本）、ACM描述符（支持SetLineCoding）[5](@ref)[50](@ref)       |

---

### **六、工程实现要点**
1. **时钟配置**  
   全速USB需48MHz时钟源（如STM32使用PLL分频），确保时序准确[4](@ref)[5](@ref)。

2. **端点初始化**  
   - 控制端点（EP0）默认启用。  
   - 数据端点需在配置阶段初始化，例如：  
     ```c
     USBD_EP_Open(EP1, USB_EP_BULK, 64); // Bulk IN端点
     USBD_EP_Open(EP2, USB_EP_BULK, 64); // Bulk OUT端点
     ```

3. **驱动兼容性**  
   Windows/Linux需依赖系统内置CDC-ACM驱动，部分旧系统需手动安装[40](@ref)[41](@ref)。

---

### **总结流程**
```
设备插入 → 复位与地址分配 → 获取描述符 → 设置配置 → 类特定配置 → 数据收发
```
通过上述流程，CDC-ACM设备实现了与主机的虚拟串口通信，适用于嵌入式调试、传感器数据传输等场景[3](@ref)[5](@ref)[15](@ref)[50](@ref)。

use futures_lite::future::block_on;
use nusb::{
    transfer::{ControlIn, ControlOut, ControlType, Recipient},
    MaybeFuture,
};

fn main() {
    // 初始化日志记录器
    env_logger::init();
    
    // 查找并获取指定VID/PID的USB设备
    let di = nusb::list_devices()
        .wait()  // 等待异步操作完成
        .unwrap()
        .find(|d| d.vendor_id() == 0x303a && d.product_id() == 0x1001)
        .expect("设备未连接，请检查USB连接");

    println!("设备信息: {di:?}");  // 打印设备详细信息

    // 打开USB设备连接
    let device = di.open().wait().unwrap();

    // Linux和macOS系统可以不声明接口直接进行控制传输
    #[cfg(any(target_os = "linux", target_os = "macos"))]
    {
        // 发送控制输出请求（Host -> Device）
        let result = block_on(device.control_out(ControlOut {
            control_type: ControlType::Vendor,  // 使用厂商特定类型
            recipient: Recipient::Device,       // 目标为设备本身
            request: 0x81,                       // 请求码
            value: 0x9999,                       // 参数值
            index: 0x9999,                       // 索引值
            data: &[1, 2, 3, 4],                 // 要发送的数据
        }));
        println!("控制输出结果: {result:?}");

        // 发送控制输入请求（Device -> Host）
        let result = block_on(device.control_in(ControlIn {
            control_type: ControlType::Vendor,
            recipient: Recipient::Device,
            request: 0x81,
            value: 0x9999,
            index: 0x9999,
            length: 256,                         // 请求返回数据的长度
        }));
        println!("控制输入结果: {result:?}");
    }

    // Windows系统必须声明接口才能进行控制传输
    // 声明使用接口0（通常为第一个接口）
    let interface = device.claim_interface(0).wait().unwrap();

    // 通过接口发送控制输出请求
    let result = block_on(interface.control_out(ControlOut {
        control_type: ControlType::Vendor,
        recipient: Recipient::Device,
        request: 0x81,
        value: 0x9999,
        index: 0x9999,
        data: &[1, 2, 3, 4],
    }));
    println!("接口控制输出结果: {result:?}");

    // 通过接口发送控制输入请求
    let result = block_on(interface.control_in(ControlIn {
        control_type: ControlType::Vendor,
        recipient: Recipient::Device,
        request: 0x81,
        value: 0x9999,
        index: 0x9999,
        length: 256,
    }));
    println!("接口控制输入结果: {result:?}");
}
