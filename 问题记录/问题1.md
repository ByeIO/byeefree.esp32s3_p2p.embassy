## 回答
根据你的错误信息和build.rs配置，链接器无法找到`link-esp32s3.x`等链接脚本文件。以下是修正方案及实现原理：

---

### **问题分析**
错误关键信息：
```sh
xtensa-esp-elf/bin/ld: cannot open linker script file link-esp32s3.x: No such file or directory
```
表明链接器在默认搜索路径中找不到链接脚本文件。虽然你的`linkall.x`和`rom_functions.x`与`build.rs`同级，但Rust/Cargo的编译工作目录可能与项目根目录不同，导致相对路径失效。

---

### **解决方案**
需要修改`build.rs`，确保链接脚本的 **绝对路径** 能被正确传递给链接器：

#### **1. 获取项目根目录路径**
在`build.rs`中通过`CARGO_MANIFEST_DIR`环境变量获取项目根目录：
```rust
use std::env;
use std::path::PathBuf;

fn main() {
    // 获取项目根目录绝对路径
    let manifest_dir = PathBuf::from(env::var("CARGO_MANIFEST_DIR").unwrap());
    
    // 构建链接脚本的完整路径
    let linkall_path = manifest_dir.join("linkall.x");
    let rom_functions_path = manifest_dir.join("rom_functions.x");
    
    // 输出链接参数（带完整路径）
    println!("cargo:rustc-link-arg-bins=-T{}", linkall_path.display());
    println!("cargo:rustc-link-arg-bins=-T{}", rom_functions_path.display());
}
```

#### **2. 验证文件位置**
确保`linkall.x`和`rom_functions.x`文件确实位于项目根目录（与`Cargo.toml`同级）。若文件在其他目录（如`ld/`子目录），需调整路径：
```rust
let linkall_path = manifest_dir.join("ld/linkall.x"); // 假设在ld目录下
```

#### **3. 补充说明**
- **环境变量**：`CARGO_MANIFEST_DIR`是Cargo自动设置的环境变量，指向`Cargo.toml`所在目录[22](@ref)。
- **跨平台兼容性**：`PathBuf::join`和`display()`方法会自动处理路径分隔符，确保Windows/macOS/Linux兼容[21](@ref)。
- **调试输出**：编译时可通过`cargo build -v`查看最终传递给链接器的完整路径是否正确。

---

### **扩展建议**
若问题仍未解决，可进一步排查：
1. **检查文件权限**：确保链接脚本文件可读。
2. **完整清理编译缓存**：运行`cargo clean`后重新编译。
3. **ESP-IDF配置**：确认是否需额外配置ESP32-S3的链接脚本（参考ESP-IDF文档中关于`-T`参数的使用）[49](@ref)。

---

通过上述修正，链接器将能正确找到脚本文件，解决`No such file or directory`错误。

```c
   Note: Master and Slave have been defined to easily understand the setup.

#include <esp_now.h>
#include <WiFi.h>
#include <esp_wifi.h> // only for esp_wifi_set_channel()

// Global copy of slave
esp_now_peer_info_t slave;
#define CHANNEL 1
#define PRINTSCANRESULTS 0
#define DELETEBEFOREPAIR 0

// Init ESP Now with fallback
void InitESPNow() {
  WiFi.disconnect();
  if (esp_now_init() == ESP_OK) {
    Serial.println("ESPNow Init Success");
  }
  else {
    Serial.println("ESPNow Init Failed");
    // Retry InitESPNow, add a counte and then restart?
    // InitESPNow();
    // or Simply Restart
    ESP.restart();
  }
}

// Scan for slaves in AP mode
void ScanForSlave() {
  int16_t scanResults = WiFi.scanNetworks(false, false, false, 300, CHANNEL); // Scan only on one channel
  // reset on each scan
  bool slaveFound = 0;
  memset(&slave, 0, sizeof(slave));

  Serial.println("");
  if (scanResults == 0) {
    Serial.println("No WiFi devices in AP Mode found");
  } else {
    Serial.print("Found "); Serial.print(scanResults); Serial.println(" devices ");
    for (int i = 0; i < scanResults; ++i) {
      // Print SSID and RSSI for each device found
      String SSID = WiFi.SSID(i);
      int32_t RSSI = WiFi.RSSI(i);
      String BSSIDstr = WiFi.BSSIDstr(i);

      if (PRINTSCANRESULTS) {
        Serial.print(i + 1);
        Serial.print(": ");
        Serial.print(SSID);
        Serial.print(" (");
        Serial.print(RSSI);
        Serial.print(")");
        Serial.println("");
      }
      delay(10);
      // Check if the current device starts with `Slave`
      if (SSID.indexOf("Slave") == 0) {
        // SSID of interest
        Serial.println("Found a Slave.");
        Serial.print(i + 1); Serial.print(": "); Serial.print(SSID); Serial.print(" ["); Serial.print(BSSIDstr); Serial.print("]"); Serial.print(" ("); Serial.print(RSSI); Serial.print(")"); Serial.println("");
        // Get BSSID => Mac Address of the Slave
        int mac[6];
        if ( 6 == sscanf(BSSIDstr.c_str(), "%x:%x:%x:%x:%x:%x",  &mac[0], &mac[1], &mac[2], &mac[3], &mac[4], &mac[5] ) ) {
          for (int ii = 0; ii < 6; ++ii ) {
            slave.peer_addr[ii] = (uint8_t) mac[ii];
          }
        }

        slave.channel = CHANNEL; // pick a channel
        slave.encrypt = 0; // no encryption

        slaveFound = 1;
        // we are planning to have only one slave in this example;
        // Hence, break after we find one, to be a bit efficient
        break;
      }
    }
  }

  if (slaveFound) {
    Serial.println("Slave Found, processing..");
  } else {
    Serial.println("Slave Not Found, trying again.");
  }

  // clean up ram
  WiFi.scanDelete();
}

// Check if the slave is already paired with the master.
// If not, pair the slave with master
bool manageSlave() {
  if (slave.channel == CHANNEL) {
    if (DELETEBEFOREPAIR) {
      deletePeer();
    }

    Serial.print("Slave Status: ");
    // check if the peer exists
    bool exists = esp_now_is_peer_exist(slave.peer_addr);
    if ( exists) {
      // Slave already paired.
      Serial.println("Already Paired");
      return true;
    } else {
      // Slave not paired, attempt pair
      esp_err_t addStatus = esp_now_add_peer(&slave);
      if (addStatus == ESP_OK) {
        // Pair success
        Serial.println("Pair success");
        return true;
      } else if (addStatus == ESP_ERR_ESPNOW_NOT_INIT) {
        // How did we get so far!!
        Serial.println("ESPNOW Not Init");
        return false;
      } else if (addStatus == ESP_ERR_ESPNOW_ARG) {
        Serial.println("Invalid Argument");
        return false;
      } else if (addStatus == ESP_ERR_ESPNOW_FULL) {
        Serial.println("Peer list full");
        return false;
      } else if (addStatus == ESP_ERR_ESPNOW_NO_MEM) {
        Serial.println("Out of memory");
        return false;
      } else if (addStatus == ESP_ERR_ESPNOW_EXIST) {
        Serial.println("Peer Exists");
        return true;
      } else {
        Serial.println("Not sure what happened");
        return false;
      }
    }
  } else {
    // No slave found to process
    Serial.println("No Slave found to process");
    return false;
  }
}

void deletePeer() {
  esp_err_t delStatus = esp_now_del_peer(slave.peer_addr);
  Serial.print("Slave Delete Status: ");
  if (delStatus == ESP_OK) {
    // Delete success
    Serial.println("Success");
  } else if (delStatus == ESP_ERR_ESPNOW_NOT_INIT) {
    // How did we get so far!!
    Serial.println("ESPNOW Not Init");
  } else if (delStatus == ESP_ERR_ESPNOW_ARG) {
    Serial.println("Invalid Argument");
  } else if (delStatus == ESP_ERR_ESPNOW_NOT_FOUND) {
    Serial.println("Peer not found.");
  } else {
    Serial.println("Not sure what happened");
  }
}

      // slave pair failed

    // No slave found to process
  }
```

```cpp
    esp_now_set_self_role(ESP_NOW_ROLE_COMBO);

    // 绑定发送回调函数
    esp_now_register_send_cb(on_data_send_callback);

    // 绑定接收回调函数
    esp_now_register_recv_cb(on_data_rec_callback);

    // 与另一个 ESP-NOW 设备（B 节点）配对以发送数据
    esp_now_add_peer(broadcast_address, ESP_NOW_ROLE_COMBO, 1, NULL, 0);

    #ifdef DEBUG
    // 发送本机地址
    Serial.print("A 节点 MAC: ");
    Serial.println(WiFi.macAddress());
    #endif
}

void loop() {
    // 检查串口是否有数据
    if (Serial.available() > 0) {
        send_serial_data();
    }
}

/*开始函数本体*/

/*
@功能:发送消息回调函数
@备注:这是一个回调函数，将在发送消息时执行。
在这种情况下，无论是否成功发送该消息，都会简单地打印出来
*/
void on_data_send_callback(uint8_t *mac_addr, uint8_t send_status) {
    #ifdef DEBUG
    Serial.print("发送状态: ");
    if (send_status == 0) {
        Serial.println("发送成功");
    } else {
        Serial.println("发送失败");
    }
    #endif
}

/*
@功能:接收消息回调函数
*/
void on_data_rec_callback(uint8_t *mac_addr, uint8_t* incoming_data, uint8_t len) {
    #ifdef DEBUG
    Serial.print("接收到来自 B 节点的消息: ");
    Serial.write(incoming_data, len);
    Serial.println();
    #endif
}

/*
@功能:发送串口接收到的数据
*/
void send_serial_data() {
    while (Serial.available() > 0) {
        usart_buffer[usart_buffer_pos] = Serial.read();
        delay(2);
        usart_buffer_pos++;
    }
    // 发送数据
    esp_now_send(broadcast_address, (uint8_t*)usart_buffer, usart_buffer_pos);

    #ifdef DEBUG
    Serial.print("发送的数据: ");
    Serial.write(usart_buffer, usart_buffer_pos);
    Serial.println();
    #endif

    usart_buffer_pos = 0;
}
```

```cpp
#include <WiFi.h>
#include <esp_now.h>

// 配置对端MAC地址（节点B的MAC地址）
uint8_t PEER_MAC[] = {0x18, 0x8b, 0x0e, 0xcc, 0x94, 0x30}; // B节点MAC:18:8b:0e:cc:94:30

// 发送回调函数
void OnDataSent(const uint8_t *mac_addr, esp_now_send_status_t status) {
  Serial.print("\r\n发送状态：");
  Serial.println(status == ESP_NOW_SEND_SUCCESS ? "成功" : "失败");
}

// 接收回调函数
void OnDataRecv(const uint8_t *mac, const uint8_t *incomingData, int len) {
  Serial.printf("\r\n来自 %02X:%02X:%02X:%02X:%02X:%02X 的消息：",
                mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
  for (int i=0; i<len; i++) {
    Serial.print((char)incomingData[i]);
  }
  Serial.println();
}

void setup() {
  Serial.begin(115200);
  WiFi.mode(WIFI_STA); // 设置WiFi为STA模式
  
  // 初始化ESP-NOW
  if (esp_now_init() != ESP_OK) {
    Serial.println("ESP-NOW初始化失败");
    ESP.restart();
  }
  
  // 注册回调函数
  esp_now_register_send_cb(OnDataSent);
  esp_now_register_recv_cb(OnDataRecv);
  
  // 配置对端设备
  esp_now_peer_info_t peerInfo;
  memcpy(peerInfo.peer_addr, PEER_MAC, 6);
  peerInfo.channel = 0;  
  peerInfo.encrypt = false;
  
  if (esp_now_add_peer(&peerInfo) != ESP_OK){
    Serial.println("添加对端设备失败");
    ESP.restart();
  }
  
  // 打印本机MAC地址
  Serial.print("当前节点MAC：");
  Serial.println(WiFi.macAddress());
  Serial.println("输入消息通过串口发送");
}

void loop() {
  // 从串口读取输入
  if (Serial.available() > 0) {
    String input = Serial.readStringUntil('\n');
    
    // 发送ESP-NOW消息
    esp_err_t result = esp_now_send(PEER_MAC, 
                                 (uint8_t *)input.c_str(), 
                                 input.length());
    
    // 处理发送结果
    if (result == ESP_OK) {
      Serial.print("已发送：");
      Serial.println(input);
    } else {
      Serial.println("发送失败");
    }
  }
}
```

根据您提供的错误日志和问题描述，以下是关于ESP32-S3出现"Peer interface is invalid"错误的分析与解决方案：

---
### **一、错误原因分析**
1. **MAC地址格式问题**  
   - 错误日志中`E (49) ESPNOW`表明对端设备添加失败，可能因MAC地址格式错误或设备未处于有效通信状态。需检查代码中的PEER_MAC数组是否符合以下要求：  
     - 需使用6字节的十六进制数组（例如`{0x18,0x8b,0x0e,0xcc,0x8a,0x24}`）  
     - 确保未混淆MAC地址的字节顺序（如将`18:8b:0e`写成`0x18,0x8b,0x0e`而非字符串形式）[1](@ref)[23](@ref)

2. **Wi-Fi模式配置冲突**  
   - 当同时使用Wi-Fi和ESP-NOW时，需设置`WIFI_AP_STA`模式（而非单纯的`WIFI_STA`）并保持设备间信道一致。若未显式设置信道，可能导致对端设备无法识别[15](@ref)[23](@ref)。

3. **固件兼容性问题**  
   - 日志显示设备使用的ESP-ROM版本为`esp32s3-20210327`（发布于2021年），可能与新版ESP-NOW协议存在兼容性问题。建议更新至最新ESP-IDF或Arduino核心库[26](@ref)[49](@ref)。

---

### **二、解决方案**
#### 1. **检查MAC地址配置**
```cpp
// 正确示例（节点B的MAC地址）

```
- **验证方法**：在双方设备上运行以下代码，确保MAC地址匹配：
  ```cpp
  Serial.print("本地MAC: ");
  Serial.println(WiFi.macAddress());
  ```

#### 2. **修改Wi-Fi模式与信道设置**
```cpp
// 在setup()中修改：
WiFi.mode(WIFI_AP_STA);  // 同时启用AP+STA模式
int32_t channel = WiFi.channel(); // 获取当前信道
// 添加对端时显式指定信道

```
- **注意**：若设备需连接WiFi网络，需先完成WiFi连接再初始化ESP-NOW，以保证信道同步[15](@ref)[23](@ref)。

#### 3. **固件与协议版本适配**
- **升级步骤**（针对Arduino环境）：
  1. 打开Arduino IDE → 工具 → 开发板管理器，更新"esp32"平台至最新版本（建议≥2.0.14）
  2. 在代码开头添加协议版本强制声明：
     ```cpp
     #define ESP_NOW_PROTOCOL_ESPNOW_V2  // 强制使用v2协议
     #include <esp_now.h>
     ```

#### 4. **硬件复位与调试**
- 若问题持续，尝试以下硬件操作：
  1. 断开所有外设（如RC522模块），仅保留必要通信电路[56](@ref)[59](@ref)
  2. 检查IO46引脚电平：确保未在启动时被拉高（可能影响刷机模式）[56](@ref)
  3. 使用USB直连而非串口转接器，避免波特率不稳定导致日志丢失
